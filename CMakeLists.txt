cmake_minimum_required(VERSION 3.24)
project(moe_expert_parallel LANGUAGES CXX)

option(MOE_ENABLE_NCCL "Build the CUDA/NCCL backend" OFF)
find_package(Threads REQUIRED)
add_library(moe_cpu STATIC cpu_backend.cpp router.cpp)
target_include_directories(moe_cpu PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(moe_cpu PUBLIC cxx_std_20)
target_link_libraries(moe_cpu PRIVATE Threads::Threads)
add_executable(moe_router main.cpp)
target_link_libraries(moe_router PRIVATE moe_cpu)

if(MOE_ENABLE_NCCL)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  find_path(NCCL_INCLUDE_DIR nccl.h REQUIRED)
  find_library(NCCL_LIBRARY nccl REQUIRED)
  target_sources(moe_router PRIVATE nccl_backend.cu)
  set_target_properties(moe_router PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
  target_compile_definitions(moe_router PRIVATE MOE_HAS_NCCL=1)
  target_include_directories(moe_router PRIVATE ${NCCL_INCLUDE_DIR})
  target_link_libraries(moe_router PRIVATE CUDA::cudart ${NCCL_LIBRARY})
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(moe_router_tests tests/router_tests.cpp)
  target_link_libraries(moe_router_tests PRIVATE moe_cpu)
  add_test(NAME moe_route_and_transport COMMAND moe_router_tests)
  add_test(NAME moe_router_cli
    COMMAND "${CMAKE_COMMAND}" "-DPROGRAM=$<TARGET_FILE:moe_router>"
            "-DNCCL_ENABLED=${MOE_ENABLE_NCCL}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake")
  set_tests_properties(moe_route_and_transport moe_router_cli PROPERTIES TIMEOUT 30)
endif()
